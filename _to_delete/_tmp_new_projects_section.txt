            var stage = document.getElementById('cfStage');
            var current = 0;
            var cards = [];

            projects.forEach(function (p, i) {
                var el = document.createElement('div');
                el.className = 'cf-card';
                el.innerHTML = '<img src="' + p.img + '" alt="' + p.title + '">' +
                    '<div class="cf-body"><h3>' + p.title + '</h3><p>' + p.teaser + '</p>' +
                    '<div class="cf-hint"><i class="fas fa-expand"></i> Click for details</div></div>';
                el.addEventListener('click', function () {
                    if (i === current) { openModal(p); } else { goTo(i); }
                });
                stage.appendChild(el);
                cards.push(el);
            });

            function layout() {
                var n = cards.length;
                var cardWidth = cards[0] ? cards[0].offsetWidth : 280;
                var stepX = cardWidth * 0.55;
                var stepZ = cardWidth * 0.42;

                cards.forEach(function (card, i) {
                    var offset = i - current;
                    if (offset > n / 2) offset -= n;
                    if (offset < -n / 2) offset += n;
                    var abs = Math.abs(offset);

                    var rot = Math.max(-55, Math.min(55, offset * -32));
                    var tx = offset * stepX;
                    var tz = -abs * stepZ;
                    var scale = Math.max(0.68, 1 - abs * 0.14);

                    card.style.marginLeft = (-cardWidth / 2) + 'px';
                    card.style.marginTop = (-(cards[0] ? cards[0].offsetHeight : 330) / 2) + 'px';
                    card.style.transform = 'translateX(' + tx + 'px) translateZ(' + tz + 'px) rotateY(' + rot + 'deg) scale(' + scale + ')';
                    card.style.filter = 'blur(' + Math.min(abs * 2.2, 6) + 'px)';
                    card.style.opacity = abs > 3 ? 0 : (1 - abs * 0.22);
                    card.style.zIndex = 100 - abs;
                    card.classList.toggle('active', offset === 0);
                });
            }

            function goTo(i) {
                var n = cards.length;
                current = ((i % n) + n) % n;
                layout();
            }

            document.getElementById('cfPrev').addEventListener('click', function () { goTo(current - 1); });
            document.getElementById('cfNext').addEventListener('click', function () { goTo(current + 1); });
            window.addEventListener('resize', layout);

            layout();

            var backdrop = document.getElementById('cfModalBackdrop');
            var modalImg = document.getElementById('cfModalImg');
            var modalTitle = document.getElementById('cfModalTitle');
            var modalBody = document.getElementById('cfModalBody');
            var modalLink = document.getElementById('cfModalLink');
            var modalLinkLabel = document.getElementById('cfModalLinkLabel');

            function openModal(p) {
                modalImg.src = p.img;
                modalImg.alt = p.title;
                modalTitle.textContent = p.title;
                modalBody.textContent = p.full;
                modalLink.href = p.link;
                modalLinkLabel.textContent = p.linkLabel;
                backdrop.classList.add('open');
            }
            function closeModal() { backdrop.classList.remove('open'); }
            document.getElementById('cfModalClose').addEventListener('click', closeModal);
            backdrop.addEventListener('click', function (e) { if (e.target === backdrop) closeModal(); });
            document.addEventListener('keydown', function (e) { if (e.key === 'Escape') closeModal(); });
        })();
